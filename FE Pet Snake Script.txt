-- Snake Bot Full Framework (Part 3: Chat Protected Input & Targeting)
local function getClosestPlayer()
    local closest, dist = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            local d = (p.Character.HumanoidRootPart.Position - hrp.Position).Magnitude
            if d < dist then closest, dist = p, d end
        end
    end
    return closest
end

local function getRandomPlayer()
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            table.insert(list, p)
        end
    end
    if #list > 0 then return list[math.random(1, #list)] end
    return nil
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed or UserInputService:GetFocusedTextBox() then return end
    
    if input.KeyCode == Enum.KeyCode.R then
        getgenv().SnakeState = "Follow"
        getgenv().IsControllingSnake = false
        print("[SNAKE AI] Mode: Following Player")
    elseif input.KeyCode == Enum.KeyCode.T then
        getgenv().SnakeState = "Roam"
        getgenv().IsControllingSnake = false
        print("[SNAKE AI] Mode: Autonomous Roaming")
    elseif input.KeyCode == Enum.KeyCode.Y then
        getgenv().CurrentTargetPlayer = getRandomPlayer()
        getgenv().SnakeState = "Random"
        getgenv().IsControllingSnake = false
        print("[SNAKE AI] Mode: Hunting Random Player")
    elseif input.KeyCode == Enum.KeyCode.U then
        getgenv().CurrentTargetPlayer = getClosestPlayer()
        getgenv().SnakeState = "Closest"
        getgenv().IsControllingSnake = false
        print("[SNAKE AI] Mode: Hunting Closest Player")
    elseif input.KeyCode == Enum.KeyCode.F then
        getgenv().IsControllingSnake = not getgenv().IsControllingSnake
        if getgenv().IsControllingSnake then
            getgenv().SnakeState = "Manual"
            human.WalkSpeed = 0 --
            human.JumpPower = 0
            print("[SNAKE AI] Mode: Manual WASD Control Active")
        else
            getgenv().SnakeState = "Follow"
            human.WalkSpeed = 16
            human.JumpPower = 50
            print("[SNAKE AI] Mode: Manual Control Terminated")
        end
    end
end)


-- Snake Bot Full Framework (Part 4: AI Brain & Head Mechanics)
local roamTime = 0
local nextRoamDirection = Vector3.new(0,0,0)
local snakeYVelocity = 0
local snakeIsGrounded = true

local function processSnakeHeadMovement(dt)
    if not hrp then return end
    local headSpeed = 14
    local currentPos = getgenv().SnakeHeadPosition
    local targetPos = currentPos
    
    if currentPos.Y > hrp.Position.Y - 2.5 then
        snakeYVelocity = snakeYVelocity - (35 * dt)
        snakeIsGrounded = false
    else
        snakeYVelocity = 0
        snakeIsGrounded = true
        targetPos = Vector3.new(currentPos.X, hrp.Position.Y - 2.5, currentPos.Z)
    end
    
    if getgenv().SnakeState == "Follow" then
        local desire = (hrp.Position + hrp.CFrame.LookVector * -3) - currentPos
        if desire.Magnitude > 3 then
            targetPos = currentPos + (desire.Unit * headSpeed * dt)
        end
        
    elseif getgenv().SnakeState == "Roam" then
        roamTime = roamTime - dt
        if roamTime <= 0 then
            roamTime = math.random(2, 5)
            nextRoamDirection = Vector3.new(math.random(-1,1), 0, math.random(-1,1)).Unit
            if math.random(1,3) == 1 and snakeIsGrounded then
                snakeYVelocity = 15 -- البوت يقفز ذاتياً
            end
        end
        targetPos = currentPos + (nextRoamDirection * (headSpeed * 0.7) * dt)
        
    elseif getgenv().SnakeState == "Random" or getgenv().SnakeState == "Closest" then
        local tPlayer = getgenv().CurrentTargetPlayer
        if tPlayer and pPlayer and tPlayer.Character and tPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local tHrp = tPlayer.Character.HumanoidRootPart
            local vec = (tHrp.Position - currentPos)
            targetPos = currentPos + (Vector3.new(vec.X, 0, vec.Z).Unit * headSpeed * dt)
            if vec.Magnitude < 10 and math.random(1,20) == 1 and snakeIsGrounded then
                snakeYVelocity = 18 -- القفز على الهدف عند الاقتراب
            end
        else
            getgenv().SnakeState = "Follow"
        end
        
    elseif getgenv().SnakeState == "Manual" and not UserInputService:GetFocusedTextBox() then
        local moveVec = Vector3.new(0,0,0)
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVec = moveVec + workspace.CurrentCamera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVec = moveVec - workspace.CurrentCamera.CFrame.LookVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVec = moveVec - workspace.CurrentCamera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVec = moveVec + workspace.CurrentCamera.CFrame.RightVector end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) and snakeIsGrounded then
            snakeYVelocity = 16 -- القفز اليدوي للثعبان
        end
        if moveVec.Magnitude > 0 then
            targetPos = currentPos + (Vector3.new(moveVec.X, 0, moveVec.Z).Unit * headSpeed * dt)
        end
    end
    
    getgenv().SnakeHeadPosition = targetPos + Vector3.new(0, snakeYVelocity * dt, 0)
end

-- Snake Bot Full Framework (Part 5: Inverse Kinematics Chain Physics)
local SegmentCFrames = {}
for i, Part in ipairs(getgenv().SnakeSegments) do
    SegmentCFrames[Part] = CFrame.new(getgenv().SnakeHeadPosition)
end

RunService.PreRender:Connect(function(dt)
    dt = math.min(dt, 0.1)
    processSnakeHeadMovement(dt) 
    
    local segments = getgenv().SnakeSegments
    local headPos = getgenv().SnakeHeadPosition
    
    for index, Part in ipairs(segments) do
        if Part and Part.Parent then
            if index == 1 then
                -- القطعة الأولى (الرأس) تتوجه وتتحرك نحو إحداثيات البوت
                local lookAtTarget = headPos + Vector3.new(0,0,0.1)
                if index < #segments and segments[index+1] then
                    lookAtTarget = segments[index+1].Position
                end
                local headCFrame = CFrame.lookAt(headPos, lookAtTarget)
                SegmentCFrames[Part] = headCFrame
            else
                local previousPart = segments[index - 1]
                local currentCFrame = SegmentCFrames[Part]
                local previousCFrame = SegmentCFrames[previousPart]
                
                local vectorToPrevious = (currentCFrame.Position - previousCFrame.Position)
                if vectorToPrevious.Magnitude > getgenv().SegmentDistance then
                    -- قفل المسافة هندسياً وسحب العقدة خلف الرأس لمنع التمدد أو التفكك البصري
                    local lockedPos = previousCFrame.Position + (vectorToPrevious.Unit * getgenv().SegmentDistance)
                    SegmentCFrames[Part] = CFrame.lookAt(lockedPos, previousCFrame.Position)
                end
            end
            
            Part.CFrame = SegmentCFrames[Part]
        end
    end
end)
